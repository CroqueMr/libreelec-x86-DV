cmake_minimum_required(VERSION 3.22)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(DVBridgeCore LANGUAGES C)
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(DVPlacebo REQUIRED IMPORTED_TARGET libplacebo>=7.372.0)
include(CheckCSourceCompiles)
set(CMAKE_REQUIRED_INCLUDES ${DVPlacebo_INCLUDE_DIRS})
check_c_source_compiles("#include <libplacebo/renderer.h>\nint main(void) { struct pl_render_params p = {0}; p.cb1_nearest_identity = 1; return !p.cb1_nearest_identity; }" CB1_POLICY_SAMPLING_FIELD)
check_c_source_compiles("#include <libplacebo/shaders/colorspace.h>\nint main(void) { struct pl_color_map_params p = {0}; p.cb1_fp32_tone_lut = 1; return !p.cb1_fp32_tone_lut; }" CB1_POLICY_TONE_FIELD)
set(CMAKE_REQUIRED_LIBRARIES PkgConfig::DVPlacebo)
check_c_source_compiles("#include <libplacebo/renderer.h>\n#include <libplacebo/shaders/sampling.h>\nint main(void) { struct pl_render_params p = {0}; struct pl_sample_filter_params s = {0}; p.cb1_fp32_scaler_lut = s.cb1_fp32_lut = 1; return pl_sample_filter_cb1_fp32_lut_format(0) != 0; }" CB1_POLICY_SCALER_ABI)
check_c_source_compiles("#include <libplacebo/renderer.h>\n#include <libplacebo/dispatch.h>\nint main(void) { struct pl_render_params p = {0}; struct pl_shader_params s = {0}; p.cb1_stable_pq = s.cb1_stable_pq = 1; pl_dispatch_mark_cb1_stable_pq(0, 1); return !p.cb1_stable_pq; }" CB1_POLICY_TRANSFER_ABI)
unset(CMAKE_REQUIRED_LIBRARIES)
if(NOT CB1_POLICY_SAMPLING_FIELD OR NOT CB1_POLICY_TONE_FIELD OR NOT CB1_POLICY_TRANSFER_ABI OR NOT CB1_POLICY_SCALER_ABI)
  message(FATAL_ERROR "Matched CB policy sampling/FP32-tone/stable-PQ/FP32-scaler libplacebo headers/library required")
endif()
unset(CMAKE_REQUIRED_INCLUDES)
pkg_check_modules(DVAvutil REQUIRED IMPORTED_TARGET libavutil>=59.12.100)
pkg_check_modules(DVAvcodec REQUIRED IMPORTED_TARGET libavcodec>=62)
pkg_check_modules(DVGles REQUIRED IMPORTED_TARGET glesv2)
pkg_check_modules(DVEgl REQUIRED IMPORTED_TARGET egl)
pkg_check_modules(DVDrm REQUIRED IMPORTED_TARGET libdrm)
add_library(dvbridge_core STATIC dvbridge_core.c dvbridge_placebo.c dvbridge_render.c
                                dvbridge_gl_frame.c dvbridge_gl_pack.c dvbridge_fel.c dvbridge_dmabuf.c
                                dvbridge_policy.c cb1_circuit.c dvbridge_creative.c)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/pack_gles300.vert" CB1_PACK_VERTEX)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/pack_gles300.frag" CB1_PACK_FRAGMENT)
foreach(variable CB1_PACK_VERTEX CB1_PACK_FRAGMENT)
  string(REPLACE "\\" "\\\\" ${variable} "${${variable}}")
  string(REPLACE "\"" "\\\"" ${variable} "${${variable}}")
  string(REPLACE "\n" "\\n\"\n\"" ${variable} "${${variable}}")
endforeach()
configure_file(dvbridge_pack_sources.h.in dvbridge_pack_sources.h @ONLY)
set_target_properties(dvbridge_core PROPERTIES POSITION_INDEPENDENT_CODE ON C_STANDARD 11)
target_include_directories(dvbridge_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_include_directories(dvbridge_core PUBLIC "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(dvbridge_core PUBLIC PkgConfig::DVPlacebo PkgConfig::DVAvutil
                                         PkgConfig::DVAvcodec PkgConfig::DVGles
                                         PkgConfig::DVEgl PkgConfig::DVDrm m)
if(CB1_ENABLE_HDR10_AI)
  target_compile_definitions(dvbridge_core PUBLIC CB1_HAS_HDR10_AI=1)
  set(CB1_L1L3_BUNDLE_CONTRACT "${CMAKE_CURRENT_SOURCE_DIR}/../models/l1l3/contract"
      CACHE PATH "Matched CB1 bundle contract directory")
  if(NOT EXISTS "${CB1_L1L3_BUNDLE_CONTRACT}/cb1_l1l3_bundle.h")
    message(FATAL_ERROR "CB1 HDR10 AI requires its generated matched bundle contract")
  endif()
  find_path(CB1_LIGHTGBM_INCLUDE_DIR LightGBM/c_api.h REQUIRED)
  find_library(CB1_LIGHTGBM_LIBRARY NAMES _lightgbm REQUIRED)
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../config/hdr10-ai-dependencies.json" ai_dependency_lock)
  string(JSON expected_api_digest GET "${ai_dependency_lock}" lightgbm c_api_sha256)
  file(SHA256 "${CB1_LIGHTGBM_INCLUDE_DIR}/LightGBM/c_api.h" actual_api_digest)
  if(NOT actual_api_digest STREQUAL expected_api_digest)
    message(FATAL_ERROR "CB1 HDR10 AI requires the pinned LightGBM 4.7.0 C API")
  endif()
  target_sources(dvbridge_core PRIVATE cb1_l1l3_model.c cb1_hdr10_features.c
                                      cb1_hdr10_extrema.c cb1_hdr10_quantiles.c
                                      cb1_hdr10_statistics.c cb1_hdr10_filter.c
                                      cb1_hdr10_ingress.c cb1_hdr10_spatial.c
                                      cb1_hdr10_temporal.c cb1_hdr10_ai.c)
  target_include_directories(dvbridge_core PRIVATE "${CB1_L1L3_BUNDLE_CONTRACT}")
  target_include_directories(dvbridge_core SYSTEM PRIVATE "${CB1_LIGHTGBM_INCLUDE_DIR}")
  target_link_libraries(dvbridge_core PUBLIC "${CB1_LIGHTGBM_LIBRARY}")
  set_source_files_properties(cb1_l1l3_model.c PROPERTIES COMPILE_OPTIONS "-ffp-contract=off;-frounding-math")
  set_source_files_properties(cb1_hdr10_features.c PROPERTIES COMPILE_OPTIONS "-ffp-contract=off;-frounding-math")
  set_source_files_properties(cb1_hdr10_quantiles.c PROPERTIES COMPILE_OPTIONS "-ffp-contract=off;-frounding-math")
  set_source_files_properties(cb1_hdr10_temporal.c PROPERTIES COMPILE_OPTIONS "-ffp-contract=off;-frounding-math")
endif()
