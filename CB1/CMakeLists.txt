cmake_minimum_required(VERSION 3.22)
project(CB1 LANGUAGES C)
option(CB1_BUILD_TESTS "Build standalone engine tests" OFF)
option(CB1_ENABLE_SANITIZERS "Enable fatal address/undefined sanitizers" OFF)
option(CB1_ENABLE_HDR10_AI "Build the matched native HDR10 L1/L3 inference candidate" OFF)
if(CB1_ENABLE_SANITIZERS)
  add_compile_options(-g -O1 -fsanitize=address,undefined -fno-sanitize-recover=undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
set(DVBRIDGE_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/src")
find_package(PkgConfig REQUIRED)
add_subdirectory(src)
include(CheckCSourceCompiles)
pkg_check_modules(CB1Avutil REQUIRED libavutil)
set(CMAKE_REQUIRED_INCLUDES ${CB1Avutil_INCLUDE_DIRS})
check_c_source_compiles("#include <libavutil/dovi_meta.h>\nint main(void) { AVDOVIDmData d = {0}; d.dvbridge_raw_magic = 0x41424456; d.dvbridge_original_length = sizeof(d.dvbridge_original_bytes); return 0; }" CB1_FFMPEG_METADATA_FIELDS)
if(NOT CB1_FFMPEG_METADATA_FIELDS)
  message(FATAL_ERROR "Matched modified FFmpeg metadata headers required")
endif()
if(CB1_BUILD_TESTS)
  enable_testing()
  add_compile_options(-UNDEBUG)
  add_subdirectory(tests)
endif()
